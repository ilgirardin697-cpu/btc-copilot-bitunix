f_getPriceValue(_htf) =>

f_crossingLevel(_curret, _level) =>
    _level > _curret and _level < _curret[1] or _level < _curret and _level > _curret[1]

alarm(_cond, _message, _htfUser, _oncePerBarClose) =>

enhanced_vwma(_series, _length, _vol) =>
    vmp = _series * _vol
    VWMA = math.sum(vmp, _length) / math.sum(_vol, _length)
    VWMA

// VAMA - Volume Adjusted Moving Average of @allanster
vama(_src, _len, _fct, _rul, _nvb) =>
    tvb = 0
    tvb := _nvb == 0 ? nz(tvb[1]) + 1 : _nvb
    tvs = _nvb == 0 ? ta.cum(volume) : math.sum(volume, _nvb)
    v2i = volume / (tvs / tvb * _fct)
    wtd = _src * v2i
    nmb = 1
    wtdSumB = 0.0
    v2iSumB = 0.0
    for i = 1 to _len * 10 by 1
        strict = _rul ? false : i == _len
        wtdSumB := wtdSumB + nz(wtd[i - 1])
        v2iSumB := v2iSumB + nz(v2i[i - 1])
        if v2iSumB >= _len or strict
            break
        nmb := nmb + 1
        nmb
    vama = (wtdSumB - (v2iSumB - _len) * _src[nmb]) / _len
    vama

// JMA - Jurik Moving Average of @everget
jma(src, length, power, phase) =>
    phaseRatio = phase < -100 ? 0.5 : phase > 100 ? 2.5 : phase / 100 + 1.5
    beta = 0.45 * (length - 1) / (0.45 * (length - 1) + 2)
    alpha = math.pow(beta, power)
    Jma = 0.0
    e0 = 0.0
    e0 := (1 - alpha) * src + alpha * nz(e0[1])
    e1 = 0.0
    e1 := (src - e0) * (1 - beta) + beta * nz(e1[1])
    e2 = 0.0
    e2 := (e0 + phaseRatio * e1 - nz(Jma[1])) * math.pow(1 - alpha, 2) + math.pow(alpha, 2) * nz(e2[1])
    Jma := e2 + nz(Jma[1])
    Jma

// ALMA - Arnaud Legoux Moving Average of @kurtsmock
enhanced_alma(_series, _length, _offset, _sigma) =>
    length = int(_length)
    numerator = 0.0
    denominator = 0.0
    m = _offset * (length - 1)
    s = length / _sigma
    for i = 0 to length - 1 by 1
        weight = math.exp(-((i - m) * (i - m)) / (2 * s * s))
        numerator := numerator + weight * _series[length - 1 - i]
        denominator := denominator + weight
        denominator
    numerator / denominator

// RSS_WMA of @RedKTrader
f_LazyLine(_data, _length) =>
    w1 = 0
    w2 = 0
    w3 = 0
    L1 = 0.0
    L2 = 0.0
    L3 = 0.0
    w = _length / 3
    if _length > 2
        w2 := math.round(w)
        w1 := math.round((_length - w2) / 2)
        w3 := int((_length - w2) / 2)
        L1 := ta.wma(_data, w1)
        L2 := ta.wma(L1, w2)
        L3 := ta.wma(L2, w3)
        L3
    else
        L3 := _data
        L3
    L3

// T3 Moving Average - Tim Tillson
f_t3(src, len) =>
    x1 = ta.ema(src, len)
    x2 = ta.ema(x1, len)
    x3 = ta.ema(x2, len)
    x4 = ta.ema(x3, len)
    x5 = ta.ema(x4, len)
    x6 = ta.ema(x5, len)
    b = 0.7
    c1 = -math.pow(b, 3)
    c2 = 3 * math.pow(b, 2) + 3 * math.pow(b, 3)
    c3 = -6 * math.pow(b, 2) - 3 * b - 3 * math.pow(b, 3)
    c4 = 1 + 3 * b + math.pow(b, 3) + 3 * math.pow(b, 2)
    c1 * x6 + c2 * x5 + c3 * x4 + c4 * x3

// Ehlers 2 Pole Super Smoother
f_2pss(_src, _length) =>
    var _a0 = math.sqrt(2) * math.pi / _length
    var _a1 = math.exp(-_a0)
    var _b0 = 2 * _a1 * math.cos(_a0)
    var _c2 = -math.pow(_a1, 2)
    var _c1 = _b0
    var _c0 = 1 - _c1 - _c2
    var _2pss = 0.0
    _src1 = nz(_src[1], _src)
    _src2 = nz(_src[2], _src1)
    _2pss := _c0 * _src + _c1 * nz(_2pss[1], _src1) + _c2 * nz(_2pss[2], _src2)
    _2pss

// VIDYA by Tushar Chande
VIDYA(src, length) =>
    valpha = 2 / (length + 1)
    vud1 = src > src[1] ? src - src[1] : 0
    vdd1 = src < src[1] ? src[1] - src : 0
    vUD = math.sum(vud1, 9)
    vDD = math.sum(vdd1, 9)
    vCMO = nz((vUD - vDD) / (vUD + vDD))
    VAR = 0.0
    VAR := nz(valpha * math.abs(vCMO) * src) + (1 - valpha * math.abs(vCMO)) * nz(VAR[1])
    VAR

// Ultimate Smoother - TASC 2024/04
UltimateSmoother(float src, int period) =>
    float a1 = math.exp(-1.414 * math.pi / period)
    float c2 = 2.0 * a1 * math.cos(1.414 * math.pi / period)
    float c3 = -a1 * a1
    float c1 = (1.0 + c2 - c3) / 4.0
    float us = src
    if bar_index >= 4
        us := (1.0 - c1) * src + (2.0 * c1 - c2) * src[1] - (c1 + c3) * src[2] + c2 * nz(us[1]) + c3 * nz(us[2])
        us
    us

ma(_type, _source, _length, _vol) =>
    switch _type
        'SMA' => ta.sma(_source, _length)
        'EMA' => ta.ema(_source, _length)
        'WMA' => ta.wma(_source, _length)
        'VWMA' => enhanced_vwma(_source, _length, _vol)
        'LSMA' => ta.linreg(_source, _length, 0)
        'ALMA' => enhanced_alma(_source, _length, 0.85, 6)
        'HMA' => ta.hma(_source, _length)
        'VAMA' => vama(_source, _length, 0.67, true, 0)
        'JMA' => jma(_source, _length, 2, 50)
        'Wilder' =>
            wild = _source
            wild := nz(wild[1]) + (_source - nz(wild[1])) / _length
            wild
        'RSS_WMA' => f_LazyLine(_source, _length)
        'D_SMA' => ta.sma(ta.sma(_source, _length), _length)
        'D_EMA' => ta.ema(ta.ema(_source, _length), _length)
        'D_WMA' => ta.wma(ta.wma(_source, _length), _length)
        'T3' => f_t3(_source, _length)
        'Ultimate Smoother' => UltimateSmoother(_source, _length)
        'Smooth LSMA' =>
            lsma = ta.linreg(_source, _length, 0)
            slsma = lsma - (lsma - ta.linreg(lsma, _length, 0))
            slsma
        '2Pole Super Smoother' => f_2pss(_source, _length)
        'D_T3' => f_t3(f_t3(_source, _length), _length)
        'VIDYA' => VIDYA(_source, _length)

f_bb(source, length, _mult) =>
    basis = ta.sma(source, length)
    dev = _mult * ta.stdev(source, length)
    upper = basis + dev
    lower = basis - dev
    avg_up_low = math.avg(upper, lower)
    diff_up_low = upper - lower
    (source - avg_up_low) / diff_up_low * 100

f_stoch(src, length, smoothFastD) =>
    ll = ta.lowest(low, length)
    hh = ta.highest(high, length)
    k = 100 * (src - ll) / (hh - ll)
    ta.sma(k, smoothFastD)
